        const AudioFX = {
            ctx: null,
            init() {
                if (!this.ctx) {
                    const AudioContext = window.AudioContext || window.webkitAudioContext;
                    if (AudioContext) this.ctx = new AudioContext();
                }
            },
            playCorrect() {
                if (!document.getElementById('setting-sfx').checked) return;
                this.init(); if (!this.ctx) return;
                const now = this.ctx.currentTime;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(523.25, now);
                osc.frequency.exponentialRampToValueAtTime(1046.50, now + 0.3);
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(now); osc.stop(now + 0.4);
            },
            playWrong() {
                if (!document.getElementById('setting-sfx').checked) return;
                this.init(); if (!this.ctx) return;
                const now = this.ctx.currentTime;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(220, now);
                osc.frequency.exponentialRampToValueAtTime(130, now + 0.3);
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(now); osc.stop(now + 0.4);
            },
            playVictory() {
                if (!document.getElementById('setting-sfx').checked) return;
                this.init(); if (!this.ctx) return;
                const now = this.ctx.currentTime;
                [523, 659, 783, 1046].forEach((f, i) => {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.frequency.value = f;
                    gain.gain.setValueAtTime(0.15, now + i * 0.1);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + i * 0.1 + 0.3);
                    osc.connect(gain); gain.connect(this.ctx.destination);
                    osc.start(now + i * 0.1); osc.stop(now + i * 0.1 + 0.3);
                });
            }
        };

        function hideAllScreens() {
            ['screen-main-menu', 'screen-story', 'screen-map', 'screen-stage', 'screen-boss', 'screen-canva-editor', 'screen-ending', 'screen-gameover'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('hidden');
            });
        }

        function showScreen(screenId) {
            hideAllScreens();
            const target = document.getElementById(screenId);
            if (target) {
                target.classList.remove('hidden');
                target.classList.add('flex');
            }
            const hud = document.getElementById('hud-bar');
            const hudMapBtn = document.getElementById('hud-map-btn');

            if (['screen-main-menu', 'screen-story', 'screen-ending', 'screen-gameover', 'screen-canva-editor'].includes(screenId)) {
                hud.classList.add('hidden');
            } else {
                hud.classList.remove('hidden');
                hudMapBtn.classList.toggle('hidden', screenId === 'screen-map');
            }
            updateHUD();
        }

        function updateHUD() {
            const container = document.getElementById('hearts-container');
            container.innerHTML = '';
            for (let i = 0; i < gameState.maxHearts; i++) {
                const s = document.createElement('span');
                s.className = 'text-base sm:text-lg';
                s.textContent = i < gameState.hearts ? '❤️' : '🖤';
                if (i >= gameState.hearts) s.classList.add('opacity-40');
                container.appendChild(s);
            }
            document.getElementById('gem-count').textContent = gameState.gemsCollected.length;
            document.getElementById('score-count').textContent = gameState.score;
        }

        function startGameStory() {
            gameState.storyIndex = 0;
            showScreen('screen-story');
            renderStorySentence();
        }

        function renderStorySentence() {
            const current = STORY_SENTENCES[gameState.storyIndex];
            document.getElementById('story-avatar').textContent = current.avatar;
            document.getElementById('story-speaker').textContent = current.speaker;
            document.getElementById('story-text').textContent = current.text;
            document.getElementById('story-progress-text').textContent = `ประโยคที่ ${gameState.storyIndex + 1}/${STORY_SENTENCES.length}`;
        }

        function nextStorySentence() {
            gameState.storyIndex++;
            if (gameState.storyIndex < STORY_SENTENCES.length) renderStorySentence();
            else goToMap();
        }

        function skipStory() { goToMap(); }

        function goToMap() {
            showScreen('screen-map');
            renderMapNodes();
        }

        function renderMapNodes() {
            const grid = document.getElementById('map-nodes-grid');
            grid.innerHTML = '';
            STAGES_DATA.forEach((stage) => {
                const isUnlocked = stage.id <= gameState.unlockedStageId;
                const isCompleted = gameState.gemsCollected.includes(stage.id);
                const card = document.createElement('div');
                card.className = `relative p-4 rounded-2xl border-2 transition-all flex flex-col justify-between ${
                    isCompleted ? 'bg-emerald-950/80 border-emerald-500 shadow-lg' : isUnlocked ? 'bg-amber-950/70 border-amber-500 hover:border-yellow-300 hover:scale-[1.02] cursor-pointer' : 'bg-slate-900/50 border-slate-700 opacity-60 cursor-not-allowed'
                }`;
                card.innerHTML = `
                    <div class="flex items-center justify-between mb-2">
                        <span class="text-3xl">${stage.icon}</span>
                        ${isCompleted ? '<span class="bg-emerald-500 text-slate-950 font-extrabold text-xs px-2.5 py-1 rounded-full">✓ ผ่านแล้ว</span>' : isUnlocked ? '<span class="bg-amber-500 text-slate-950 font-bold text-xs px-2 py-0.5 rounded-full">ปลดล็อกแล้ว</span>' : '<span class="text-slate-500 text-lg">🔒</span>'}
                    </div>
                    <div>
                        <div class="text-xs text-amber-300/80 font-semibold">ด่านที่ ${stage.id}</div>
                        <h4 class="font-bold text-base text-slate-100">${stage.title}</h4>
                    </div>
                    <div class="mt-3 pt-2 border-t border-slate-800 flex justify-between items-center text-xs">
                        <span class="text-slate-400">รางวัล: 💎</span>
                        ${isUnlocked ? `<button onclick="enterStage(${stage.id})" class="btn-rpg-wood text-xs px-3 py-1 rounded-lg text-yellow-100 font-bold">${isCompleted ? 'เล่นซ้ำ' : 'เข้าผจญภัย'}</button>` : '<span class="text-slate-500">ล็อกอยู่</span>'}
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function enterStage(stageId) {
            gameState.currentStageId = stageId;
            if (stageId === 12) { enterBossStage(); return; }
            const stage = STAGES_DATA.find(s => s.id === stageId);
            if (!stage) return;

            showScreen('screen-stage');
            document.getElementById('screen-stage').style.backgroundImage = `linear-gradient(to bottom, rgba(15, 23, 42, 0.85), rgba(15, 23, 42, 0.95)), url('${stage.bgImg}')`;

            document.getElementById('stage-icon').textContent = stage.icon;
            document.getElementById('stage-number-text').textContent = `ด่านที่ ${stage.id}`;
            document.getElementById('stage-title-text').textContent = stage.title;
            document.getElementById('stage-npc-avatar').textContent = stage.npcIcon;
            document.getElementById('stage-npc-name').textContent = stage.npcName;
            document.getElementById('stage-dialogue-speaker').textContent = `${stage.npcName} กล่าวว่า:`;
            document.getElementById('stage-dialogue-text').textContent = `"${stage.dialogue}"`;
            document.getElementById('stage-question-text').textContent = stage.question;

            const container = document.getElementById('stage-choices-container');
            container.innerHTML = '';
            const prefixes = ['A', 'B', 'C', 'D'];
            stage.choices.forEach((choice, index) => {
                const btn = document.createElement('button');
                btn.className = 'w-full text-left bg-slate-800/90 hover:bg-slate-700/90 border border-slate-600 hover:border-amber-400 p-3 rounded-xl transition text-sm flex items-start space-x-2 shadow-md';
                btn.onclick = () => handleAnswerChoice(index);
                btn.innerHTML = `<span class="bg-amber-600/40 text-amber-300 border border-amber-500/50 rounded px-2 py-0.5 font-bold text-xs mt-0.5">${prefixes[index]}</span><span class="text-slate-100 flex-grow">${choice}</span>`;
                container.appendChild(btn);
            });
        }

        function handleAnswerChoice(selectedIndex) {
            const stage = STAGES_DATA.find(s => s.id === gameState.currentStageId);
            if (!stage) return;
            const isCorrect = selectedIndex === stage.correctIndex;

            if (isCorrect) {
                AudioFX.playCorrect();
                gameState.score += 10;
                if (!gameState.gemsCollected.includes(stage.id)) gameState.gemsCollected.push(stage.id);
                if (stage.id === gameState.unlockedStageId && gameState.unlockedStageId < 12) gameState.unlockedStageId++;
                updateHUD();

                showResultModal({
                    isSuccess: true,
                    title: "เก่งมาก! ตอบถูกแล้ว!",
                    message: `ยอดเยี่ยม! รับไปเลย +10 คะแนน และได้ 💎 อัญมณีแห่งปัญญาประจำด่าน ${stage.title}`,
                    explanation: stage.explanation,
                    btnText: stage.id === 12 ? "ปราบจอมปีศาจ!" : "ไปด่านต่อไป ➔",
                    onConfirm: () => {
                        if (stage.id < 12) enterStage(stage.id + 1);
                        else triggerVictoryEnding();
                    }
                });
                if (document.getElementById('setting-glow').checked) confetti({ particleCount: 40, spread: 60, origin: { y: 0.7 } });
            } else {
                AudioFX.playWrong();
                gameState.hearts--;
                updateHUD();

                if (gameState.hearts <= 0) {
                    showResultModal({
                        isSuccess: false, title: "หัวใจหมดแล้ว...", message: "ไม่เป็นไรนะ ลองเริ่มใหม่อีกครั้ง!", explanation: stage.explanation, btnText: "ดูหน้า Game Over",
                        onConfirm: () => showScreen('screen-gameover')
                    });
                } else {
                    showResultModal({
                        isSuccess: false, title: "ยังไม่ถูกต้องนะ", message: `ลองคิดดูใหม่อีกทีนะ (เหลือ ❤️ ${gameState.hearts} ดวง)`, explanation: stage.explanation, btnText: "ลองใหม่อีกครั้ง 🔄"
                    });
                }
            }
        }

        function enterBossStage() {
            gameState.currentStageId = 12;
            gameState.bossDialogueIndex = 0;
            showScreen('screen-boss');
            document.getElementById('boss-dialogue-text').textContent = `"${BOSS_DIALOGUES[0]}"`;
            document.getElementById('boss-question-section').classList.add('hidden');
            document.getElementById('boss-next-dialogue-btn').classList.remove('hidden');
        }

        function advanceBossDialogue() {
            gameState.bossDialogueIndex++;
            if (gameState.bossDialogueIndex < BOSS_DIALOGUES.length) {
                document.getElementById('boss-dialogue-text').textContent = `"${BOSS_DIALOGUES[gameState.bossDialogueIndex]}"`;
            } else {
                document.getElementById('boss-next-dialogue-btn').classList.add('hidden');
                document.getElementById('boss-dialogue-text').textContent = `"ตอบคำถามนี้ให้ได้ ถ้าอยากช่วยเจ้าหญิง!"`;
                const bossStage = STAGES_DATA.find(s => s.id === 12);
                document.getElementById('boss-question-text').textContent = bossStage.question;

                const container = document.getElementById('boss-choices-container');
                container.innerHTML = '';
                const prefixes = ['A', 'B', 'C', 'D'];
                bossStage.choices.forEach((choice, index) => {
                    const btn = document.createElement('button');
                    btn.className = 'w-full text-left bg-red-950/80 hover:bg-red-900 border border-red-700 hover:border-amber-400 p-3 rounded-xl transition text-sm flex items-start space-x-2 shadow-lg';
                    btn.onclick = () => handleBossAnswerChoice(index);
                    btn.innerHTML = `<span class="bg-amber-600 text-amber-100 rounded px-2 py-0.5 font-bold text-xs mt-0.5">${prefixes[index]}</span><span class="text-slate-100 flex-grow">${choice}</span>`;
                    container.appendChild(btn);
                });
                document.getElementById('boss-question-section').classList.remove('hidden');
            }
        }

        function handleBossAnswerChoice(selectedIndex) {
            const bossStage = STAGES_DATA.find(s => s.id === 12);
            if (selectedIndex === bossStage.correctIndex) {
                document.getElementById('boss-knight-avatar').classList.add('animate-attack-knight');
                setTimeout(() => document.getElementById('boss-demon-avatar').classList.add('animate-hit-demon'), 300);
                setTimeout(() => {
                    document.getElementById('boss-knight-avatar').classList.remove('animate-attack-knight');
                    document.getElementById('boss-demon-avatar').classList.remove('animate-hit-demon');
                    handleAnswerChoice(selectedIndex);
                }, 700);
            } else {
                handleAnswerChoice(selectedIndex);
            }
        }

        /* ----------------------------------------------------
           CANVA EDITOR STUDIO LOGIC
        ---------------------------------------------------- */
        function openCanvaEditor() {
            showScreen('screen-canva-editor');
            renderEditorTabs();
            loadStageToEditor(gameState.editorStageIndex);
        }

        function renderEditorTabs() {
            const container = document.getElementById('editor-stage-tabs-list');
            container.innerHTML = '';
            STAGES_DATA.forEach((stage, idx) => {
                const btn = document.createElement('button');
                btn.className = `px-3 py-1.5 rounded-xl font-bold text-xs whitespace-nowrap transition border ${
                    idx === gameState.editorStageIndex ? 'bg-purple-600 border-purple-300 text-white shadow-lg' : 'bg-slate-800 border-slate-700 text-slate-300 hover:bg-slate-700'
                }`;
                btn.textContent = `${stage.icon} ด่านที่ ${stage.id}`;
                btn.onclick = () => {
                    gameState.editorStageIndex = idx;
                    renderEditorTabs();
                    loadStageToEditor(idx);
                };
                container.appendChild(btn);
            });
        }

        function loadStageToEditor(idx) {
            const stage = STAGES_DATA[idx];
            if (!stage) return;

            document.getElementById('edit-input-title').value = stage.title;
            document.getElementById('edit-input-icon').value = stage.icon;
            document.getElementById('edit-input-npc-name').value = stage.npcName;
            document.getElementById('edit-input-npc-icon').value = stage.npcIcon;
            document.getElementById('edit-input-bg').value = stage.bgImg;
            document.getElementById('edit-input-dialogue').value = stage.dialogue;
            document.getElementById('edit-input-question').value = stage.question;

            stage.choices.forEach((c, i) => {
                document.getElementById(`edit-input-choice-${i}`).value = c;
            });

            document.getElementById('edit-input-correct').value = stage.correctIndex;
            document.getElementById('edit-input-explanation').value = stage.explanation;

            updateEditorPreview();
        }

        function updateEditorPreview() {
            const idx = gameState.editorStageIndex;
            STAGES_DATA[idx].title = document.getElementById('edit-input-title').value;
            STAGES_DATA[idx].icon = document.getElementById('edit-input-icon').value;
            STAGES_DATA[idx].npcName = document.getElementById('edit-input-npc-name').value;
            STAGES_DATA[idx].npcIcon = document.getElementById('edit-input-npc-icon').value;
            STAGES_DATA[idx].bgImg = document.getElementById('edit-input-bg').value;
            STAGES_DATA[idx].dialogue = document.getElementById('edit-input-dialogue').value;
            STAGES_DATA[idx].question = document.getElementById('edit-input-question').value;
            STAGES_DATA[idx].choices[0] = document.getElementById('edit-input-choice-0').value;
            STAGES_DATA[idx].choices[1] = document.getElementById('edit-input-choice-1').value;
            STAGES_DATA[idx].choices[2] = document.getElementById('edit-input-choice-2').value;
            STAGES_DATA[idx].choices[3] = document.getElementById('edit-input-choice-3').value;
            STAGES_DATA[idx].correctIndex = parseInt(document.getElementById('edit-input-correct').value);
            STAGES_DATA[idx].explanation = document.getElementById('edit-input-explanation').value;

            document.getElementById('edit-preview-stage-tag').textContent = `ด่านที่ ${STAGES_DATA[idx].id}`;
            document.getElementById('editor-preview-bg').style.backgroundImage = `linear-gradient(to bottom, rgba(15, 23, 42, 0.6), rgba(15, 23, 42, 0.9)), url('${STAGES_DATA[idx].bgImg}')`;
            document.getElementById('editor-prev-icon-title').textContent = `${STAGES_DATA[idx].icon} ${STAGES_DATA[idx].title}`;
            document.getElementById('editor-prev-npc-icon').textContent = STAGES_DATA[idx].npcIcon;
            document.getElementById('editor-prev-npc-name').textContent = STAGES_DATA[idx].npcName;
            document.getElementById('editor-prev-dialogue').textContent = `"${STAGES_DATA[idx].dialogue}"`;
            document.getElementById('editor-prev-question').textContent = STAGES_DATA[idx].question;
            document.getElementById('editor-prev-a').textContent = STAGES_DATA[idx].choices[0];
            document.getElementById('editor-prev-b').textContent = STAGES_DATA[idx].choices[1];
        }

        function triggerVictoryEnding() {
            showScreen('screen-ending');
            AudioFX.playVictory();
            document.getElementById('end-score').textContent = gameState.score;
            document.getElementById('end-gems').textContent = `${gameState.gemsCollected.length}/12`;
            if (document.getElementById('setting-glow').checked) {
                confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } });
            }
        }

        let modalConfirmCallback = null;
        function showResultModal({ isSuccess, title, message, explanation, btnText, onConfirm }) {
            const modal = document.getElementById('modal-result');
            const card = document.getElementById('modal-result-card');
            document.getElementById('result-icon').textContent = isSuccess ? '🎉' : '❌';
            document.getElementById('result-title').textContent = title;
            document.getElementById('result-title').className = `text-2xl font-bold mb-2 ${isSuccess ? 'text-emerald-400' : 'text-red-400'}`;
            document.getElementById('result-message').textContent = message;

            const expBox = document.getElementById('result-explanation-box');
            if (explanation) {
                expBox.classList.remove('hidden');
                document.getElementById('result-explanation-text').textContent = explanation;
            } else { expBox.classList.add('hidden'); }

            document.getElementById('result-action-btn').textContent = btnText || 'ตกลง';
            modalConfirmCallback = onConfirm;

            modal.classList.remove('hidden');
            setTimeout(() => { card.classList.remove('scale-95', 'opacity-0'); card.classList.add('scale-100', 'opacity-100'); }, 10);
        }

        function closeResultModal() {
            const modal = document.getElementById('modal-result');
            const card = document.getElementById('modal-result-card');
            card.classList.remove('scale-100', 'opacity-100'); card.classList.add('scale-95', 'opacity-0');
            setTimeout(() => {
                modal.classList.add('hidden');
                if (modalConfirmCallback) { modalConfirmCallback(); modalConfirmCallback = null; }
            }, 150);
        }

        function openModal(id) { document.getElementById(id).classList.remove('hidden'); }
        function closeModal(id) { document.getElementById(id).classList.add('hidden'); }

        function restartGame() {
            gameState.score = 0; gameState.hearts = 3; gameState.gemsCollected = [];
            gameState.currentStageId = 1; gameState.unlockedStageId = 1; gameState.storyIndex = 0;
            updateHUD(); goToMap();
        }

        function returnToMainMenu() {
            gameState.score = 0; gameState.hearts = 3; gameState.gemsCollected = [];
            gameState.currentStageId = 1; gameState.unlockedStageId = 1;
            updateHUD(); showScreen('screen-main-menu');
        }
